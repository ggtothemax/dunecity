cmake_minimum_required(VERSION 3.15)
project(DuneCity VERSION 1.0.0 LANGUAGES NONE)
get_filename_component(GAME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_SIZEOF_VOID_P 8)
set(CPACK_VERBATIM_VARIABLES ON)
set(CPACK_PACKAGE_NAME "${PROJECT_NAME}")
set(CPACK_PACKAGE_VENDOR "Dune City Team")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_RESOURCE_FILE_LICENSE "${GAME_ROOT}/COPYING")
set(CPACK_RESOURCE_FILE_README "${GAME_ROOT}/README.md")
include("${GAME_ROOT}/cmake/WindowsInstaller.cmake")
# This non-executable payload exercises packaging only and is never uploaded.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dunecity.exe" "Installer packaging fixture")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/dunecity.exe" DESTINATION .)
install(FILES "${CPACK_RESOURCE_FILE_README}" DESTINATION . RENAME README)
include(CPack)
